cmake_minimum_required(VERSION 3.28)
project(hw2vr VERSION 0.4.1 LANGUAGES CXX RC)

# HomeworldRM.exe is 32-bit (docs/FINDINGS.md), so the proxy must be too.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "HomeworldRM.exe is 32-bit: configure for Win32 (cmake --preset win32).")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Static CRT: nothing extra to ship next to the game.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# OpenVR SDK v2.15.6 (third_party/openvr/README.md). Header only at build time; openvr_api.dll is loaded at runtime.
set(OPENVR_DIR ${CMAKE_SOURCE_DIR}/third_party/openvr)

add_library(hw2vr_proxy SHARED
    src/proxy/dllmain.cpp
    src/proxy/exports.cpp
    src/proxy/gl_hooks.cpp
    src/proxy/gl_api.cpp
    src/stereo/view_override.cpp
    src/hud/hud_layer.cpp
    src/vr/openvr_session.cpp
    src/vr/vr_frame.cpp
    src/common/log.cpp
    src/common/config.cpp
    src/common/hotkeys.cpp
    src/proxy/opengl32.def   # generated by tools/gen_exports.py
    src/proxy/version.rc
)
set_target_properties(hw2vr_proxy PROPERTIES OUTPUT_NAME opengl32)
target_include_directories(hw2vr_proxy PRIVATE src)
target_include_directories(hw2vr_proxy SYSTEM PRIVATE ${OPENVR_DIR}/include)
target_compile_definitions(hw2vr_proxy PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(hw2vr_proxy PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /permissive- /utf-8 /external:W0>)
# No incremental-link thunks: exports should point straight at the stubs/hooks (overlays inline-hook them).
target_link_options(hw2vr_proxy PRIVATE /INCREMENTAL:NO)
# Ship the 32-bit OpenVR runtime client next to the proxy (tools/install.ps1 copies both).
add_custom_command(TARGET hw2vr_proxy POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${OPENVR_DIR}/bin/win32/openvr_api.dll $<TARGET_FILE_DIR:hw2vr_proxy>)

# Loads the proxy the same way the game does (static import, found in the exe directory). Run: build/win32/<cfg>/hw2vr_smoke.exe
add_executable(hw2vr_smoke tests/proxy_smoke.cpp)
target_compile_definitions(hw2vr_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(hw2vr_smoke PRIVATE opengl32 gdi32 user32)
add_dependencies(hw2vr_smoke hw2vr_proxy)

# Pure per-eye camera maths (no GL / OpenVR).
add_executable(hw2vr_math_test tests/math_test.cpp)
target_include_directories(hw2vr_math_test PRIVATE src)
target_compile_options(hw2vr_math_test PRIVATE /W4 /permissive- /utf-8)
